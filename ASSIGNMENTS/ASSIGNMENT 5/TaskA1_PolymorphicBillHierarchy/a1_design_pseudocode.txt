ALGORITHM ComputeDailyBillTotal(bills, count)
INPUT:
    bills : ARRAY[0 .. count - 1] OF POINTER TO Bill
    count : INTEGER
OUTPUT:
    grandTotal : REAL

BEGIN
    grandTotal <- 0.0
    FOR index <- 0 TO count - 1 DO
        IF bills[index] != NULL THEN
            CALL bills[index]->print()
            billAmount <- bills[index]->total()
            grandTotal <- grandTotal + billAmount
        END IF
    END FOR
    RETURN grandTotal
END ALGORITHM
